   677→    /**
   678→     * Small wrapper around the Mail.tm REST API.
   679→     * - Calls the official API first; if the browser blocks it (CORS / network),
   680→     *   transparently retries once through the same-origin relay endpoint.
   681→     * - Auto-retries rate-limited (429) responses with a short backoff.
   682→     */
   683→    async function api(path, { method = 'GET', body, auth = false, timeout, contentType } = {}) {
   684→      const call = (base) => {
   685→        const headers = { Accept: 'application/json' };
   686→        if (contentType) headers['Content-Type'] = contentType;
   687→        else if (body !== undefined) headers['Content-Type'] = 'application/json';
   688→        if (auth && state.account && state.account.token) headers.Authorization = 'Bearer ' + state.account.token;
   689→        return fetchWithTimeout(base + path, {
   690→          method,
   691→          headers,
   692→          body: body !== undefined ? JSON.stringify(body) : undefined,
   693→        }, timeout);
   694→      };
   695→
   696→      const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
   697→      let res = null;
   698→
   699→      for (let attempt = 0; attempt < 3; attempt++) {
   700→        try {
   701→          res = await call(API_BASE);
   702→        } catch (err) {
   703→          if (err instanceof TypeError && API_BASE === DIRECT_API && RELAY_API) {
   704→            console.warn('[AR TEMPMAIL] direct Mail.tm API unreachable — switching to relay endpoint');
   705→            API_BASE = RELAY_API;
   706→            showToast('Using secure relay for the Mail.tm API', 'info');
   707→            continue; // retry immediately through the relay
   708→          } else if (err instanceof TypeError) {
   709→            throw new Error('Network error — check your connection.');
   710→          } else {
   711→            throw err;
   712→          }
   713→        }
   714→        if (res.status === 429 && attempt < 2) {
   715→          await sleep(1500 * (attempt + 1)); // rate limited → back off and retry
   716→          continue;
   717→        }
   718→        break;
   719→      }
   720→
   721→      if (res.status === 204) return null;
   722→
   723→      let data = null;
   724→      try { data = await res.json(); } catch (e) { data = null; }
   725→
   726→      if (!res.ok) {
   727→        const err = new Error(
   728→          (data && (data['hydra:description'] || data.detail || data.message)) ||
   729→          ('Request failed (' + res.status + ')')
   730→        );
   731→        err.status = res.status;
   732→        throw err;
   733→      }
   734→      return data;
   735→    }

  1120→    /* ---------------- Core: generate a new temp email ---------------- */
  1121→    async function getNewEmail(triggerBtn = null) {
  1122→      if (state.generating) return;
  1123→      state.generating = true;
  1124→      if (triggerBtn) showSpinner(triggerBtn);
  1125→      stopPolling();
  1126→      stopTimer();
  1127→      setEmailLoading(true);
  1128→      setStatusBadge('GENERATING', 'loading');
  1129→
  1130→      try {
  1131→        /* 1) Fetch available public domain */
  1132→        const domainsData = await api('/domains?page=1');
  1133→        const domains = ((domainsData && domainsData['hydra:member']) || domainsData || [])
  1134→          .filter((d) => d && d.isActive && !d.isPrivate && d.domain)
  1135→          .map((d) => d.domain);
  1136→        if (!domains.length) throw new Error('No mail domains available right now.');
  1137→        const domain = domains[Math.floor(Math.random() * domains.length)];
  1138→
  1139→        /* 2) Create account (retry on address collision) */
  1140→        let account = null;
  1141→        for (let attempt = 0; attempt < 4 && !account; attempt++) {
  1142→          const address = (randomLocalPart() + '@' + domain).toLowerCase();
  1143→          const password = randString(16);
  1144→          try {
  1145→            const created = await api('/accounts', { method: 'POST', body: { address, password } });
  1146→            account = { id: created.id, address: created.address || address, password };
  1147→          } catch (err) {
  1148→            if (err.status === 422 || err.status === 400) continue; // address taken → retry
  1149→            throw err;
  1150→          }
  1151→        }
  1152→        if (!account) throw new Error('Could not create a mailbox. Please try again.');
  1153→
  1154→        /* 3) Get auth token */
  1155→        const tokenData = await api('/token', {
  1156→          method: 'POST',
  1157→          body: { address: account.address, password: account.password },
  1158→        });
  1159→        account.token = tokenData.token;
  1160→
  1161→        /* 4) Switch session */
  1162→        state.account = account;
  1163→        state.messages = [];
  1164→        state.seenIds = new Set();
  1165→        state.hiddenIds = new Set();
  1166→        state.firstLoadDone = false;
  1167→        state.searchQuery = '';
  1168→        el.inboxSearch.value = '';
  1169→        state.generating = false;
  1170→
  1171→        persistActiveSession();
  1172→        setEmailLoading(false);
  1173→        setStatusBadge('ACTIVE', 'active');

  1412→    function sanitizeHTML(dirty = '') {
  1413→      const doc = new DOMParser().parseFromString(dirty, 'text/html');
  1414→      doc.querySelectorAll('script, style, iframe, frame, frameset, object, embed, link, meta, base, form, input, button, textarea, select')
  1415→        .forEach((n) => n.remove());
  1416→      doc.querySelectorAll('*').forEach((node) => {
  1417→        for (const attr of Array.from(node.attributes || [])) {
  1418→          const name = attr.name.toLowerCase();
  1419→          if (name.startsWith('on')) { node.removeAttribute(attr.name); continue; }
  1420→          if ((name === 'href' || name === 'src' || name === 'xlink:href' || name === 'srcset' || name === 'action') &&
  1421→              /^\s*(javascript|vbscript|data:text\/html)/i.test(attr.value)) {
  1422→            node.removeAttribute(attr.name);
  1423→          }
  1424→        }
  1425→        if (node.tagName === 'A') {
  1426→          node.setAttribute('target', '_blank');
  1427→          node.setAttribute('rel', 'noopener noreferrer nofollow');
  1428→        }
  1429→      });
  1430→      return (doc.body && doc.body.innerHTML) || '';
  1431→    }
