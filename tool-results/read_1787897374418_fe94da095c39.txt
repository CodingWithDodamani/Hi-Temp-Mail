   811→      const diff = Math.max(0, Date.now() - then);
   812→      const s = Math.floor(diff / 1000);

   841→    /** Cryptographically-strong character picker (falls back to Math.random). */
   842→    function pickRandom(chars) {
   843→      if (globalThis.crypto && crypto.getRandomValues) {
   844→        // Rejection sampling keeps the distribution uniform for any charset size
   845→        const limit = Math.floor(256 / chars.length) * chars.length;
   846→        const buf = new Uint8Array(1);
   847→        do { crypto.getRandomValues(buf); } while (buf[0] >= limit);
   848→        return chars[buf[0] % chars.length];
   849→      }
   850→      return chars[Math.floor(Math.random() * chars.length)];
   851→    }

   901→    function extractOtp(...parts) {

   928→    /**
   929→     * Small wrapper around the Mail.tm REST API.
   930→     * - Calls the official API first; if the browser blocks it (CORS / network),
   931→     *   transparently retries once through the same-origin relay endpoint.
   932→     * - Auto-retries rate-limited (429) responses with a short backoff.
   933→     */
   934→    async function api(path, { method = 'GET', body, auth = false, timeout, contentType } = {}) {
   935→      const call = (base) => {
   936→        const headers = { Accept: 'application/json' };
   937→        if (contentType) headers['Content-Type'] = contentType;
   938→        else if (body !== undefined) headers['Content-Type'] = 'application/json';
   939→        if (auth && state.account && state.account.token) headers.Authorization = 'Bearer ' + state.account.token;
   940→        return fetchWithTimeout(base + path, {
   941→          method,
   942→          headers,
   943→          body: body !== undefined ? JSON.stringify(body) : undefined,
   944→        }, timeout);
   945→      };
   946→
   947→      const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
   948→      let res = null;
   949→
   950→      for (let attempt = 0; attempt < 3; attempt++) {
   951→        try {
   952→          res = await call(API_BASE);
   953→        } catch (err) {
   954→          if (err instanceof TypeError && API_BASE === DIRECT_API && RELAY_API) {
   955→            console.warn('[AR TEMPMAIL] direct Mail.tm API unreachable — switching to relay endpoint');
   956→            API_BASE = RELAY_API;
   957→            showToast('Using secure relay for the Mail.tm API', 'info');
   958→            continue; // retry immediately through the relay
   959→          } else if (err instanceof TypeError) {
   960→            throw new Error('Network error — check your connection.');
   961→          } else {
   962→            throw err;
   963→          }
   964→        }
   965→        if (res.status === 429 && attempt < 2) {
   966→          await sleep(1500 * (attempt + 1)); // rate limited → back off and retry
   967→          continue;
   968→        }
   969→        break;
   970→      }
   971→
   972→      if (res.status === 204) return null;
   973→
   974→      let data = null;
   975→      try { data = await res.json(); } catch (e) { data = null; }
   976→
   977→      if (!res.ok) {
   978→        const err = new Error(
   979→          (data && (data['hydra:description'] || data.detail || data.message)) ||
   980→          ('Request failed (' + res.status + ')')
   981→        );
   982→        err.status = res.status;
   983→        throw err;
   984→      }
   985→      return data;
   986→    }

  1429→    /* ---------------- Core: generate a new temp email ---------------- */
  1430→    async function getNewEmail(triggerBtn = null) {
  1431→      if (state.generating) return;
  1432→      state.generating = true;
  1433→      if (triggerBtn) showSpinner(triggerBtn);
  1434→      stopPolling();
  1435→      stopTimer();
  1436→      setEmailLoading(true);
  1437→      setStatusBadge('GENERATING', 'loading');
  1438→
  1439→      try {
  1440→        /* 1) Fetch available public domain */
  1441→        const domainsData = await api('/domains?page=1');
  1442→        const domains = ((domainsData && domainsData['hydra:member']) || domainsData || [])
  1443→          .filter((d) => d && d.isActive && !d.isPrivate && d.domain)
  1444→          .map((d) => d.domain);
  1445→        if (!domains.length) throw new Error('No mail domains available right now.');
  1446→        const domain = domains[Math.floor(Math.random() * domains.length)];
  1447→
  1448→        /* 2) Create account (retry on address collision) */
  1449→        let account = null;
  1450→        for (let attempt = 0; attempt < 4 && !account; attempt++) {
  1451→          const address = (randomLocalPart() + '@' + domain).toLowerCase();
  1452→          const password = randString(16);
  1453→          try {
  1454→            const created = await api('/accounts', { method: 'POST', body: { address, password } });
  1455→            account = { id: created.id, address: created.address || address, password };
  1456→          } catch (err) {
  1457→            if (err.status === 422 || err.status === 400) continue; // address taken → retry
  1458→            throw err;
  1459→          }
  1460→        }
  1461→        if (!account) throw new Error('Could not create a mailbox. Please try again.');
  1462→
  1463→        /* 3) Get auth token */
  1464→        const tokenData = await api('/token', {
  1465→          method: 'POST',
  1466→          body: { address: account.address, password: account.password },
  1467→        });
  1468→        account.token = tokenData.token;
  1469→
  1470→        /* 4) Switch session */
  1471→        state.account = account;
  1472→        state.messages = [];
  1473→        state.seenIds = new Set();
  1474→        state.hiddenIds = new Set();
  1475→        state.firstLoadDone = false;
  1476→        state.searchQuery = '';
  1477→        el.inboxSearch.value = '';
  1478→        state.generating = false;
  1479→
  1480→        persistActiveSession();
  1481→        setEmailLoading(false);
  1482→        setStatusBadge('ACTIVE', 'active');
  1483→        renderInbox();
  1484→        updateSaveBtn();
  1485→        startTimer(TIMER_SECONDS);
  1486→        startPolling();
  1487→        checkInbox(false);
  1488→        showToast('New email generated!', 'success');
