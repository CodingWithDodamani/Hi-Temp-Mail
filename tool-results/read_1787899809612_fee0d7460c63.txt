   903→    /** Cryptographically-strong character picker (falls back to Math.random). */
   904→    function pickRandom(chars) {
   905→      if (globalThis.crypto && crypto.getRandomValues) {
   906→        // Rejection sampling keeps the distribution uniform for any charset size
   907→        const limit = Math.floor(256 / chars.length) * chars.length;
   908→        const buf = new Uint8Array(1);
   909→        do { crypto.getRandomValues(buf); } while (buf[0] >= limit);
   910→        return chars[buf[0] % chars.length];
   911→      }
   912→      return chars[Math.floor(Math.random() * chars.length)];
   913→    }

   962→    /** Heuristic OTP / verification-code extractor. */
   963→    function extractOtp(...parts) {
   964→      const hay = parts.filter(Boolean).join(' ').replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' ');
   965→      const keyword = /(?:\b(?:code|otp|pin|passcode|password|verification|verify|confirm|token|security|authenticate)\b[^0-9]{0,24}?)(\d{3}[\s-]?\d{3}|\d{4,8})\b/i;
   966→      let m = hay.match(keyword);
   967→      if (m) return m[1].replace(/[\s-]/g, '');
   968→      m = hay.match(/\b(\d{3}[\s-]\d{3})\b/);
   969→      if (m) return m[1].replace(/[\s-]/g, '');
   970→      m = hay.match(/\b(\d{6})\b/);
   971→      if (m) return m[1];
   972→      m = hay.match(/\b(\d{4,8})\b/);
   973→      return m ? m[1] : null;
   974→    }

   976→    /** Safe fetch with timeout error handling. */
   977→    async function fetchWithTimeout(url, options = {}, timeout = 15000) {
   978→      const controller = new AbortController();
   979→      const timer = setTimeout(() => controller.abort(), timeout);
   980→      try {
   981→        return await fetch(url, { ...options, signal: controller.signal });
   982→      } catch (err) {
   983→        if (err && err.name === 'AbortError') throw new Error('Request timed out. Check your connection.');
   984→        throw err; // keep raw TypeError so the CORS/network fallback can detect it
   985→      } finally {
   986→        clearTimeout(timer);
   987→      }

   996→    async function api(path, { method = 'GET', body, auth = false, timeout, contentType } = {}) {
   997→      const call = (base) => {

  1003→          method,
  1004→          headers,
  1005→          body: body !== undefined ? JSON.stringify(body) : undefined,
  1006→        }, timeout);
  1007→      };
  1008→
  1009→      const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
  1010→      let res = null;
  1011→
  1012→      for (let attempt = 0; attempt < 3; attempt++) {
  1013→        try {
  1014→          res = await call(API_BASE);
  1015→        } catch (err) {
  1016→          if (err instanceof TypeError && API_BASE === DIRECT_API && RELAY_API) {
  1017→            console.warn('[AR TEMPMAIL] direct Mail.tm API unreachable — switching to relay endpoint');
  1018→            API_BASE = RELAY_API;
  1019→            showToast('Using secure relay for the Mail.tm API', 'info');
  1020→            continue; // retry immediately through the relay
  1021→          } else if (err instanceof TypeError) {
  1022→            throw new Error('Network error — check your connection.');
  1023→          } else {
  1024→            throw err;
  1025→          }
  1026→        }
  1027→        if (res.status === 429 && attempt < 2) {
  1028→          await sleep(1500 * (attempt + 1)); // rate limited → back off and retry
  1029→          continue;
  1030→        }
  1031→        break;
  1032→      }
  1033→
  1034→      if (res.status === 204) return null;
  1035→
  1036→      let data = null;
  1037→      try { data = await res.json(); } catch (e) { data = null; }
  1038→
  1039→      if (!res.ok) {
  1040→        const err = new Error(
  1041→          (data && (data['hydra:description'] || data.detail || data.message)) ||
  1042→          ('Request failed (' + res.status + ')')
  1043→        );
  1044→        err.status = res.status;
  1045→        throw err;
  1046→      }
  1047→      return data;
  1048→    }

  1510→    async function getNewEmail(triggerBtn = null) {
  1511→      if (state.generating) return;
  1512→      state.generating = true;
  1513→      if (triggerBtn) showSpinner(triggerBtn);
  1514→      stopPolling();
  1515→      stopTimer();
  1516→      setEmailLoading(true);
  1517→      setStatusBadge('GENERATING', 'loading');
  1518→
  1519→      try {
  1520→        /* 1) Fetch available public domain */
  1521→        const domainsData = await api('/domains?page=1');
  1522→        const domains = ((domainsData && domainsData['hydra:member']) || domainsData || [])
  1523→          .filter((d) => d && d.isActive && !d.isPrivate && d.domain)
  1524→          .map((d) => d.domain);
  1525→        if (!domains.length) throw new Error('No mail domains available right now.');
  1526→        const domain = domains[Math.floor(Math.random() * domains.length)];
  1527→
  1528→        /* 2) Create account (retry on address collision) */
  1529→        let account = null;
  1530→        for (let attempt = 0; attempt < 4 && !account; attempt++) {
  1531→          const address = (randomLocalPart() + '@' + domain).toLowerCase();
  1532→          const password = randString(16);
  1533→          try {
  1534→            const created = await api('/accounts', { method: 'POST', body: { address, password } });
  1535→            account = { id: created.id, address: created.address || address, password };
  1536→          } catch (err) {
  1537→            if (err.status === 422 || err.status === 400) continue; // address taken → retry
  1538→            throw err;
  1539→          }
  1540→        }
  1541→        if (!account) throw new Error('Could not create a mailbox. Please try again.');
  1542→
  1543→        /* 3) Get auth token */
  1544→        const tokenData = await api('/token', {
  1545→          method: 'POST',
  1546→          body: { address: account.address, password: account.password },
  1547→        });
  1548→        account.token = tokenData.token;
  1549→
  1550→        /* 4) Switch session */
  1551→        state.account = account;
  1552→        state.messages = [];
  1553→        state.seenIds = new Set();
  1554→        state.hiddenIds = new Set();
  1555→        state.firstLoadDone = false;
  1556→        state.searchQuery = '';
  1557→        el.inboxSearch.value = '';
  1558→        state.generating = false;
  1559→
  1560→        persistActiveSession();
  1561→        setEmailLoading(false);
  1562→        setStatusBadge('ACTIVE', 'active');
  1563→        renderInbox();
  1564→        updateSaveBtn();
  1565→        startTimer(TIMER_SECONDS);
  1566→        startPolling();
  1567→        checkInbox(false);
  1568→        showToast('New email generated!', 'success');
  1569→      } catch (err) {
  1570→        state.generating = false;
  1571→        console.error('[AR TEMPMAIL] getNewEmail failed:', err);
  1572→        setStatusBadge('OFFLINE', 'error');
  1573→        setEmailError(err.message || 'Failed to generate email.');
  1574→        showToast(err.message || 'Failed to generate email.', 'error');
  1575→      } finally {
  1576→        state.generating = false;
  1577→        if (triggerBtn) hideSpinner(triggerBtn);
  1578→      }
  1579→    }
